module memory_controller (
  input  wire        clk,                       //clock
  input  wire        rst_n,                     //async active-low reset
  input  wire [15:0] cpu_addr,                  //cpu address
  input  wire [15:0] cpu_wdata,                 //cpu write data
  output reg  [15:0] cpu_rdata,                 //cpu read data
  input  wire        cpu_r_en,                  //cpu read request
  input  wire        cpu_w_en,                  //cpu write request
  output wire        ready,                     //controller ready when idle and no pending read
  output reg         read_valid,                //read data valid pulse
  output reg         write_done,                //write done pulse
  output reg  [15:0] mem_addr,                  //memory address
  output reg  [15:0] mem_wdata,                 //memory write data
  input  wire [15:0] mem_rdata,                 //memory read data
  output reg         mem_r_en,                  //memory read enable
  output reg         mem_w_en                   //memory write enable
);

localparam S_IDLE  = 2'd0;                      //idle state
localparam S_WRITE = 2'd1;                      //write state
localparam S_READ  = 2'd2;                      //read state

reg [1:0] state, state_n;                       //fsm state and next-state
reg [1:0] cnt, cnt_n; //                         //tiny latency counter and next
reg pend_read_valid, pend_read_valid_n;         //queued read flag and next
reg [15:0] pend_read_addr, pend_read_addr_n;    //queued read address and next

// Signals _n
reg [15:0] mem_addr_n, mem_wdata_n, cpu_rdata_n; //next-cycle versions of outputs
reg mem_r_en_n, mem_w_en_n, read_valid_n, write_done_n; //next-cycle control pulses

assign ready = (state == S_IDLE) && !pend_read_valid; //ready only in idle with no pending read

always @* begin                                  //combinational next-state/output logic
  // Default assignments
  state_n = state;                                //hold state by default
  cnt_n = cnt;                                    //hold counter
  mem_addr_n = mem_addr;                          //hold outputs
  mem_wdata_n = mem_wdata;
  cpu_rdata_n = cpu_rdata;
  mem_r_en_n = 1'b0;                              //deassert controls by default
  mem_w_en_n = 1'b0;
  read_valid_n = 1'b0;
  write_done_n = 1'b0;
  pend_read_valid_n = pend_read_valid;            //hold pending read info
  pend_read_addr_n = pend_read_addr;

  case (state)
    S_IDLE: begin                                 //idle: accept new requests
      cnt_n = 2'd1;                               //model 1-cycle latency
      if(cpu_w_en && cpu_r_en) begin              //both read and write in same cycle
        pend_read_valid_n = 1'b1;                 //queue read
        pend_read_addr_n = cpu_addr;              //store read address
        mem_addr_n = cpu_addr;                    //start write now
        mem_wdata_n = cpu_wdata;
          mem_w_en_n = 1'b1;
        state_n = S_WRITE;                        //go to write
      end else if(cpu_w_en) begin                 //write only
        mem_addr_n = cpu_addr;
        mem_wdata_n = cpu_wdata;
        mem_w_en_n = 1'b1;
        state_n = S_WRITE;
      end else if(cpu_r_en) begin                 //read only
        mem_addr_n = cpu_addr;
        mem_r_en_n = 1'b1;
        state_n = S_READ;
      end
    end

    S_WRITE: begin                                //write state
      if(cnt == 2'd0) begin                       //latency elapsed
        write_done_n = 1'b1;                      //pulse write_done
        if(pend_read_valid) begin                 //serve queued read
          mem_addr_n = pend_read_addr;
          pend_read_valid_n = 1'b0;
          mem_r_en_n = 1'b1;
          state_n = S_READ;
          cnt_n = 2'd1;                           //apply read latency
        end else begin
          state_n = S_IDLE;                       //return to idle
        end
      end else cnt_n = cnt - 2'd1;                //count down
    end

    S_READ: begin                                 //read state
      if(cnt == 2'd0) begin                       //latency elapsed
        cpu_rdata_n = mem_rdata;                  //capture memory data
        read_valid_n = 1'b1;                      //pulse read_valid
        state_n = S_IDLE;                         //return to idle
      end else cnt_n = cnt - 2'd1;                //count down
    end
  endcase
end

always @(posedge clk or negedge rst_n) begin      //sequential registers with async reset
  if(!rst_n) begin
    state <= S_IDLE;                              //reset to idle
    cnt <= 2'd0;                                  //clear counter
    pend_read_valid <= 1'b0;                      //no pending read
    pend_read_addr <= 16'b0;                      //clear queued address
    mem_addr <= 16'b0;                            //clear outputs
    mem_wdata <= 16'b0;
    cpu_rdata <= 16'b0;
    mem_r_en <= 1'b0;
    mem_w_en <= 1'b0;
    read_valid <= 1'b0;
    write_done <= 1'b0;
  end else begin
    state <= state_n;                             //update state
    cnt <= cnt_n;                                 //update counter
    pend_read_valid <= pend_read_valid_n;         //update pending flags
    pend_read_addr <= pend_read_addr_n;
    mem_addr <= mem_addr_n;                       //update outputs
    mem_wdata <= mem_wdata_n;
    cpu_rdata <= cpu_rdata_n;
    mem_r_en <= mem_r_en_n;
    mem_w_en <= mem_w_en_n;
    read_valid <= read_valid_n;
    write_done <= write_done_n;
  end
end

endmodule

module simple_memory (                            //single-port ram with sync write and registered read
  input wire clk, rst_n,                          //clock and async active-low reset
  input wire [15:0] mem_addr,                     //address from controller
  input wire [15:0] mem_wdata,                    //write data from controller
  output reg [15:0] mem_rdata,                    //registered read data to controller
  input wire mem_r_en, mem_w_en                   //read/write enables
);
 
reg [15:0] mem [0:1023];                          //1k x 16-bit memory array
reg [9:0] rd_addr_q;                              //registered read address (10 bits for 1k depth)
 
always @(posedge clk or negedge rst_n) begin       //sync logic with async reset
  if(!rst_n) begin                                 //on reset clear read addr and data
    rd_addr_q <= 0;                                //reset read address
    mem_rdata <= 0;                                //reset read data
  end else begin
    if(mem_w_en) mem[mem_addr[9:0]] <= mem_wdata;  //sync write on clock edge
    if(mem_r_en) rd_addr_q <= mem_addr[9:0];       //capture read address when enabled
    mem_rdata <= mem[rd_addr_q];                   //output data from previous captured address
  end
end
 
endmodule
module simple_cpu (                                        //simple driver fsm: write->read->rw
  input wire clk, rst_n,                                   //clock and async active-low reset
  output reg [15:0] cpu_addr, cpu_wdata,                   //address and write data to controller
  output reg cpu_r_en, cpu_w_en,                           //read/write enables to controller
  input wire [15:0] cpu_rdata,                             //read data from controller
  input wire ready,                                        //controller ready
  input wire read_valid, write_done                        //status pulses from controller
);

localparam S_IDLE  = 2'd0;                                 //idle state
localparam S_WRITE = 2'd1;                                 //write state
localparam S_READ  = 2'd2;                                 //read state
localparam S_RW    = 2'd3;                                 //simultaneous r/w state

reg [1:0] state, state_n;                                  //fsm state and next-state

always @* begin
  state_n = state;                                         //hold by default
  cpu_r_en = 1'b0;                                         //deassert controls by default
  cpu_w_en = 1'b0;
  cpu_addr = 16'b0;                                        //default address/data
  cpu_wdata = 16'b0;

  case(state)
    S_IDLE: begin                                          //wait until controller is ready
      if (ready) begin
        cpu_addr = 16'h10;                                 //target address
        cpu_wdata = 16'hBEEF;                              //data to write
        cpu_w_en = 1'b1;                                   //issue write
        state_n = S_WRITE;                                 //go write
      end
    end

    S_WRITE: begin                                         //keep write asserted until done
      cpu_addr = 16'h10;
      cpu_wdata = 16'hBEEF;
      cpu_w_en = 1'b1;
      if (write_done) begin
        state_n = S_READ;                                  //then read same address
      end
    end

    S_READ: begin                                           
      cpu_r_en = 1'b1;
      if (read_valid) begin
        state_n = S_RW;                                    //next do simultaneous r/w
      end
    end

    S_RW: begin                                            //drive read+write together
      cpu_addr = 16'h20;                                   //new address for write
      cpu_wdata = 16'h1234;
      cpu_w_en = 1'b1;
      cpu_r_en = 1'b1;
      if (write_done) begin                                //write expected to finish first
        state_n = S_IDLE;                                  //return to idle
      end
    end
  endcase
end

always @(posedge clk or negedge rst_n) begin               //sequential state register
  if(!rst_n) state <= S_IDLE;                              //reset to idle
  else state <= state_n;                                   //update state
end

endmodule 
module top_system(input wire clk, input wire rst_n); //top-level integration of cpu, controller, memory
  wire [15:0] cpu_addr, cpu_wdata, cpu_rdata;       //cpu<->ctrl data/address bus
  wire cpu_r_en, cpu_w_en, ready, read_valid, write_done; //handshake/control signals
  wire [15:0] mem_addr, mem_wdata, mem_rdata;       //ctrl<->mem data/address bus
  wire mem_r_en, mem_w_en;                           //memory enables

  simple_cpu cpu(                                    //traffic generator (write->read->rw)
    .clk(clk), .rst_n(rst_n),
    .cpu_addr(cpu_addr), .cpu_wdata(cpu_wdata),
    .cpu_r_en(cpu_r_en), .cpu_w_en(cpu_w_en),
    .cpu_rdata(cpu_rdata),
    .ready(ready), .read_valid(read_valid), .write_done(write_done)
  );

  memory_controller ctrl(                            //arbiter/fsm with write priority
    .clk(clk), .rst_n(rst_n),
    .cpu_addr(cpu_addr), .cpu_wdata(cpu_wdata),
    .cpu_rdata(cpu_rdata),
    .cpu_r_en(cpu_r_en), .cpu_w_en(cpu_w_en),
    .ready(ready), .read_valid(read_valid), .write_done(write_done),
    .mem_addr(mem_addr), .mem_wdata(mem_wdata),
    .mem_rdata(mem_rdata),
    .mem_r_en(mem_r_en), .mem_w_en(mem_w_en)
  );

  simple_memory mem(                                 //1k×16 single-port ram (sync write, reg read)
    .clk(clk), .rst_n(rst_n),
    .mem_addr(mem_addr), .mem_wdata(mem_wdata),
    .mem_rdata(mem_rdata),
    .mem_r_en(mem_r_en), .mem_w_en(mem_w_en)
  );
endmodule

timescale 1ns/1ps                         //timescale for simulation units/precision
module tb_memory_controller;              //simple testbench to drive memory_controller
  reg clk, rst_n;                         //clock and async active-low reset
  reg  [15:0] cpu_addr, cpu_wdata;        //cpu address and write data
  reg         cpu_r_en, cpu_w_en;         //cpu read/write enables
  wire [15:0] cpu_rdata;                  //cpu read data from dut
  wire ready, read_valid, write_done;     //handshake/status pulses from dut
  wire [15:0] mem_addr, mem_wdata;        //memory-side address/write data
  reg  [15:0] mem_rdata;                  //memory-side read data (driven by tb)
  wire mem_r_en, mem_w_en;                //memory-side enables from dut

  memory_controller u_ctrl (              //device under test instantiation
    .clk(clk), .rst_n(rst_n),
    .cpu_addr(cpu_addr), .cpu_wdata(cpu_wdata),
    .cpu_rdata(cpu_rdata),
    .cpu_r_en(cpu_r_en), .cpu_w_en(cpu_w_en),
    .ready(ready), .read_valid(read_valid), .write_done(write_done),
    .mem_addr(mem_addr), .mem_wdata(mem_wdata),
    .mem_rdata(mem_rdata),
    .mem_r_en(mem_r_en), .mem_w_en(mem_w_en)
  );

  initial clk = 0; always #5 clk = ~clk;  //100mhz clock generator (10ns period)
  initial begin rst_n = 0; #25 rst_n = 1; end //async reset release after 25ns

  initial begin                            //main stimulus sequence
    cpu_addr = 0; cpu_wdata = 0; cpu_r_en = 0; cpu_w_en = 0; mem_rdata = 0; //init tb signals
    @(posedge rst_n);                      //wait for reset deassertion

    // Write Operation
    $display("\n[TB] Starting WRITE operation at time %0t", $time); //log start
    @(posedge clk);
    cpu_addr = 16'h0010; cpu_wdata = 16'hBEEF; cpu_w_en = 1; //issue write
    @(posedge clk); cpu_w_en = 0;             //deassert write request
    #10; // Wait for write_done to assert     //simple delay (could also wait on posedge write_done)
    if (write_done)
      $display("[TB] WRITE SUCCESS: data 0x%h written to address 0x%h at time %0t", 16'hBEEF, 16'h0010, $time); //expect pass
    else
      $display("[TB] WRITE FAILED at time %0t", $time); //flag failure

    //read Operation
    $display("\n[TB] Starting READ operation at time %0t", $time); //log start
    mem_rdata = 16'hBEEF;                   //provide memory read data for dut
    @(posedge clk); cpu_addr = 16'h0010; cpu_r_en = 1; //issue read
    @(posedge clk); cpu_r_en = 0;           //deassert read request
    #10; // Wait for read_valid to assert    //simple delay (could also wait on posedge read_valid)
    if (read_valid && (cpu_rdata == 16'hBEEF))
      $display("[TB] READ SUCCESS: data 0x%h read from address 0x%h at time %0t", cpu_rdata, 16'h0010, $time); //expect pass
    else
      $display("[TB] READ FAILED at time %0t", $time); //flag failure

    //simultaneous R/W operation
    $display("\n[TB] Starting SIMULTANEOUS R/W operation at time %0t", $time); //log start
    @(posedge clk); 
    cpu_addr = 16'h0020; cpu_wdata = 16'h1234; cpu_w_en = 1; cpu_r_en = 1; //assert both enables same cycle
    @(posedge clk); cpu_w_en = 0; cpu_r_en = 0; //deassert both

    //assume write completes first, then read begins
    #10; // Wait for write_done to assert     //write should have priority
    if (write_done)
      $display("[TB] SIMUL RW: WRITE of 0x1234 to 0x0020 completed successfully."); //expect pass
    else
      $display("[TB] SIMUL RW: WRITE FAILED."); //flag failure
      
    // Wait for read to complete
    #10;                                      //allow time before read completes
    mem_rdata = 16'h1234;                     //provide memory data for deferred read
    #10;
    if (read_valid && (cpu_rdata == 16'h1234))
      $display("[TB] SIMUL RW: READ of 0x1234 from 0x0020 completed successfully."); //expect pass
    else
      $display("[TB] SIMUL RW: READ FAILED."); //flag failure
      
    #100 $display("\n[TB] Simulation finished"); $finish; //end simulation
  end
endmodule

`timescale 1ns/1ps                          //timescale for simulation
module tb_simple_memory;                    //simple testbench for simple_memory
  reg clk, rst_n;                           //clock and async active-low reset
  reg  [15:0] mem_addr, mem_wdata;          //address and write data driven by tb
  reg         mem_r_en, mem_w_en;           //read/write enables driven by tb
  wire [15:0] mem_rdata;                    //read data returned from dut
 
  simple_memory u_mem (                     //dut instantiation
    .clk(clk),
    .rst_n(rst_n),
    .mem_addr(mem_addr),
    .mem_wdata(mem_wdata),
    .mem_rdata(mem_rdata),
    .mem_r_en(mem_r_en),
    .mem_w_en(mem_w_en)
  );
 
  initial clk = 0; always #5 clk = ~clk;    //100mhz clock (10ns period)
  initial begin rst_n = 0; #25 rst_n = 1; end //release reset after 25ns
 
  initial begin                             //main stimulus
    mem_addr = 0; mem_wdata = 0; mem_r_en = 0; mem_w_en = 0; //init tb signals
    @(posedge rst_n);                       //wait for reset deassertion
    
    //Write test 
    $display("\n[MEM TB] Starting WRITE test at time %0t", $time); //log start
    mem_addr = 16'h000A; mem_wdata = 16'hCAFE; mem_w_en = 1;  //issue write
    @(posedge clk); 
    mem_w_en = 0; 
    $display("[MEM TB] WRITE @0x%h <= 0x%h at time %0t", mem_addr, mem_wdata, $time); //write info
 
    //Read test
    $display("\n[MEM TB] Starting READ test at time %0t", $time); //log start
    mem_addr = 16'h000A; mem_r_en = 1;         //request read
    @(posedge clk); 
    mem_r_en = 0;
    @(posedge clk); // Wait for the read data to be available //registered read path needs 1 cycle
    
    if (mem_rdata == 16'hCAFE)
      $display("[MEM TB] READ SUCCESS: data 0x%h read from address 0x%h at time %0t", mem_rdata, mem_addr, $time); //expect pass
    else
      $display("[MEM TB] READ FAILED: Expected 0xCAFE, Got 0x%h at time %0t", mem_rdata, $time); //flag failure
 
    #50 $display("\n[MEM TB] Simulation finished"); $finish; //end simulation
  end
Endmodule                                      //testbench end

`timescale 1ns/1ps
module tb_simple_cpu;
  reg clk, rst_n;
  wire [15:0] cpu_addr, cpu_wdata;
  wire cpu_r_en, cpu_w_en;
  reg  [15:0] cpu_rdata;
  reg ready, read_valid, write_done;

  simple_cpu u_cpu (
    .clk(clk), .rst_n(rst_n),
    .cpu_addr(cpu_addr), .cpu_wdata(cpu_wdata),
    .cpu_r_en(cpu_r_en), .cpu_w_en(cpu_w_en),
    .cpu_rdata(cpu_rdata),
    .ready(ready),
    .read_valid(read_valid),
    .write_done(write_done)
  );

  initial clk = 0; always #5 clk = ~clk;
  initial begin rst_n = 0; #25 rst_n = 1; end

  initial begin
    ready = 0; read_valid = 0; write_done = 0; cpu_rdata = 0;
    @(posedge rst_n);

    // Test scenario 1: Write operation
    $display("\n[TB] Starting WRITE test at time %0t", $time);
    @(posedge clk);
    ready = 1; // Assert ready to allow CPU to start
    @(posedge clk);
    ready = 0;
    // Check if CPU sent a write request
    if (cpu_w_en) $display("[TB] CPU sent a WRITE request for addr=0x%h, data=0x%h at time %0t", cpu_addr, cpu_wdata, $time);
    
    @(posedge clk);
    write_done = 1; // Simulate the memory controller asserting write_done
    @(posedge clk);
    write_done = 0;
    $display("[TB] Memory Controller asserted WRITE_DONE at time %0t", $time);

    // Test scenario 2: Read operation
    $display("\n[TB] Starting READ test at time %0t", $time);
    @(posedge clk);
    ready = 1; // Assert ready to allow CPU to start
    @(posedge clk);
    ready = 0;
    // Check if CPU sent a read request
    if (cpu_r_en) $display("[TB] CPU sent a READ request for addr=0x%h at time %0t", cpu_addr, $time);

    @(posedge clk);
    cpu_rdata = 16'hBEEF; // Provide dummy data from memory
    read_valid = 1; // Simulate the memory controller asserting read_valid
    @(posedge clk);
    read_valid = 0;
    $display("[TB] Memory Controller asserted READ_VALID with data 0x%h at time %0t", cpu_rdata, $time);

    // Test scenario 3: Simultaneous R/W
    $display("\n[TB] Starting SIMULTANEOUS R/W test at time %0t", $time);
    @(posedge clk);
    ready = 1;
    @(posedge clk);
    ready = 0;
    if (cpu_w_en && cpu_r_en) $display("[TB] CPU sent SIMULTANEOUS R/W request at time %0t", $time);
    
    #100 $display("\n[TB] Simulation finished"); $finish;
  end
endmodule

`timescale 1ns/1ps
module tb_top_system;                               //top-level system testbench
  reg clk, rst_n;                                   //clock and async active-low reset
  
  top_system uut (.clk(clk), .rst_n(rst_n));        //dut instantiation

  initial clk = 0; always #5 clk = ~clk;            //100mhz clock
  initial begin rst_n = 0; #25 rst_n = 1; end       //release reset after 25ns

  initial begin
    $dumpfile("system.vcd");                        //vcd dump for waves
    $dumpvars(0, tb_top_system);                    //dump all under tb

    @(posedge rst_n);                               //wait for reset deassert
    $display("[TB] Starting simulation...");        //start banner
    
    @(posedge uut.cpu_w_en);                        //wait cpu write request
    $display("\n[TB] Detected WRITE request at time %0t. CPU wants to write 0x%h to 0x%h.", $time, uut.cpu_wdata, uut.cpu_addr);
    
    @(posedge uut.write_done);                      //wait write completion
    $display("[TB] WRITE operation completed at time %0t.", $time);
    
    @(posedge uut.cpu_r_en);                        //wait cpu read request
    $display("\n[TB] Detected READ request at time %0t. CPU wants to read from 0x%h.", $time, uut.cpu_addr);

    @(posedge uut.read_valid);                      //wait read completion
    $display("[TB] READ operation completed at time %0t. CPU received data 0x%h.", $time, uut.cpu_rdata);
    
    @(posedge uut.cpu_w_en);                        //wait simultaneous r/w start
    $display("\n[TB] Detected SIMULTANEOUS R/W request at time %0t.", $time);
    @(posedge uut.write_done);                      //write finishes first
    $display("[TB] SIMUL RW: WRITE portion completed at time %0t.", $time);
    @(posedge uut.read_valid);                      //then deferred read completes
    $display("[TB] SIMUL RW: READ portion completed at time %0t.", $time);

    #500;                                           //settle and finish
    $display("\n[TB] Simulation finished");
    $finish;
  end
endmodule
